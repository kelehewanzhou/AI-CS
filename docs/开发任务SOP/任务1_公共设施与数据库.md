CREATE TABLE users (

CREATE TABLE sessions (

**表9: learning_events**
```sql
CREATE TABLE learning_events (
    event_id bigint GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    user_id uuid NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
    event_type text NOT NULL CHECK (event_type IN ('view_node', 'view_edge', 'start_exercise')),
    node_id varchar(16) REFERENCES knowledge_nodes(node_id) ON DELETE RESTRICT,
    edge_id varchar(16) REFERENCES knowledge_edges(edge_id) ON DELETE RESTRICT,
    question_id varchar(32) REFERENCES exercises(question_id) ON DELETE RESTRICT,
    created_at timestamptz DEFAULT clock_timestamp() NOT NULL,
    CONSTRAINT event_resource_check CHECK (
        (event_type = 'view_node' AND node_id IS NOT NULL AND edge_id IS NULL AND question_id IS NULL) OR
        (event_type = 'view_edge' AND edge_id IS NOT NULL AND node_id IS NULL AND question_id IS NULL) OR
        (event_type = 'start_exercise' AND question_id IS NOT NULL AND node_id IS NULL AND edge_id IS NULL)
    )
);

CREATE INDEX idx_events_user_node ON learning_events(user_id, node_id, event_type);
CREATE INDEX idx_events_user_time ON learning_events(user_id, created_at DESC, event_id DESC);
```

### 错误格式（§3.1）

```python
from pydantic import BaseModel
from typing import Optional, Dict, Any

class ErrorResponse(BaseModel):
    code: str  # VALIDATION_ERROR, UNAUTHENTICATED, FORBIDDEN, NOT_FOUND, CONFLICT, RATE_LIMITED, INTERNAL_ERROR, UNAVAILABLE
    message: str
    request_id: str
    details: Optional[Dict[str, Any]] = None
```

8个固定错误码：
- `VALIDATION_ERROR` (422): 类型/枚举/长度错误、缺少字段
- `UNAUTHENTICATED` (401): 未登录、会话无效
- `FORBIDDEN` (403): 角色不符、Origin/CSRF不通过
- `NOT_FOUND` (404): 资源不存在或已下线
- `CONFLICT` (409): 并发冲突
- `RATE_LIMITED` (429): 限流，附Retry-After
- `INTERNAL_ERROR` (500): 服务内部异常
- `UNAVAILABLE` (503): 数据库不可用

### backend/common/errors.py

```python
from fastapi import Request, status
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from pydantic import BaseModel
from typing import Optional, Dict, Any
import uuid

class ErrorResponse(BaseModel):
    code: str
    message: str
    request_id: str
    details: Optional[Dict[str, Any]] = None

class AppException(Exception):
    """业务异常基类"""
    def __init__(self, code: str, message: str, status_code: int, details: Optional[Dict] = None):
        self.code = code
        self.message = message
        self.status_code = status_code
        self.details = details

def get_request_id(request: Request) -> str:
    """从请求头获取或生成request_id"""
    request_id = request.headers.get("x-request-id")
    if not request_id:
        request_id = uuid.uuid4().hex
    return request_id

async def app_exception_handler(request: Request, exc: AppException):
    return JSONResponse(
        status_code=exc.status_code,
        content={
            "code": exc.code,
            "message": exc.message,
            "request_id": get_request_id(request),
            "details": exc.details
        }
    )

async def validation_exception_handler(request: Request, exc: RequestValidationError):
    return JSONResponse(
        status_code=422,
        content={
            "code": "VALIDATION_ERROR",
            "message": "请求参数验证失败",
            "request_id": get_request_id(request),
            "details": {"errors": exc.errors()}
        }
    )

async def generic_exception_handler(request: Request, exc: Exception):
    return JSONResponse(
        status_code=500,
        content={
            "code": "INTERNAL_ERROR",
            "message": "服务器内部错误",
            "request_id": get_request_id(request)
        }
    )
```

### backend/api/routers/health.py

```python
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session
from sqlalchemy import text
from common.database import get_db
from common.errors import AppException
from common.config import settings

router = APIRouter()

@router.get("/live")
def liveness():
    """进程存活检查，无鉴权"""
    return {"data": {"status": "ok"}, "request_id": "internal"}

@router.get("/ready")
def readiness(db: Session = Depends(get_db)):
    """就绪检查：数据库+迁移+种子"""
    try:
        # 检查数据库连接
        db.execute(text("SELECT 1"))
        
        # 检查迁移版本表是否存在
        result = db.execute(text(
            "SELECT EXISTS (SELECT FROM information_schema.tables WHERE table_name = 'alembic_version')"
        ))
        if not result.scalar():
            raise AppException("UNAVAILABLE", "数据库迁移未执行", 503)
        
        # 检查是否有成功的导入记录
        result = db.execute(text(
            "SELECT EXISTS (SELECT FROM import_runs WHERE status = 'succeeded' LIMIT 1)"
        ))
        if not result.scalar():
            raise AppException("UNAVAILABLE", "未导入种子数据", 503)
        
        return {"data": {"status": "ready"}, "request_id": "internal"}
    except AppException:
        raise
    except Exception as e:
        raise AppException("UNAVAILABLE", f"数据库检查失败: {str(e)}", 503)
```

### database/migrations/env.py（Alembic配置）

```python
from logging.config import fileConfig
from sqlalchemy import engine_from_config, pool
from alembic import context
import sys
from pathlib import Path

# 添加backend到路径
sys.path.insert(0, str(Path(__file__).parent.parent.parent / "backend"))

from common.database import Base
from common.config import settings

# 导入所有模型（后续任务会添加）
# from auth.models import User, Session
# ...

config = context.config
config.set_main_option("sqlalchemy.url", settings.database_url)

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata

def run_migrations_online():
    connectable = engine_from_config(
        config.get_section(config.config_ini_section),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(connection=connection, target_metadata=target_metadata)
        with context.begin_transaction():
            context.run_migrations()

run_migrations_online()
```

## 要使用的字段和接口

### 题型规则（§3.5）

**单选题（choice）**：
- options: 2-6个`{key: "A"-"F", text: "..."}`
- answer_key: JSON字符串，必须属于选项
- 提交答案：字符串，trim后转大写

**判断题（judge）**：
- options: 必须null
- answer_key: JSON boolean
- 提交答案：严格boolean，不接受字符串或数字

**简答题（short_answer）**：
- P0拒绝导入和提交

```python
# 1. 检查UNIQUE(user_id, client_submission_id)
# 2. 存在且question_id和标准化答案相同 → 返回原记录（200）
```

### backend/exercise/service.py

```python
from uuid import UUID
from typing import Any
from sqlalchemy.orm import Session
from sqlalchemy import text, func
from common.errors import AppException
from .repository import ExerciseRepository
from .models import Exercise, ExerciseAttempt

class ExerciseService:
    def __init__(self, db: Session):
        self.db = db
        self.repo = ExerciseRepository(db)
    
    def list_exercises(
        self, 
        node_id: Optional[str] = None,
        question_id: Optional[str] = None,
        limit: int = 10,
        offset: int = 0
    ) -> tuple[List[ExerciseStudent], int]:
        """列出题目学生视图"""
        exercises, total = self.repo.list_active(node_id, question_id, limit, offset)
        
        student_views = []
        for ex in exercises:
            related = self.repo.get_related_nodes(ex.question_id)
            student_views.append(ExerciseStudent(
                question_id=ex.question_id,
                type=ex.type,
                content=ex.content,
                options=ex.options,
                difficulty=ex.difficulty,
                related_node_ids=[r.node_id for r in related],
                is_active=ex.is_active
            ))
        
        return student_views, total
    
    def submit_attempt(
        self,
        user_id: UUID,
        question_id: str,
        answer: Any,
        client_submission_id: UUID
    ) -> tuple[AttemptView, bool]:
        """
        提交答题，返回(AttemptView, created)
        created=True表示首次创建，False表示幂等返回
        """
        # 1. 检查题目存在
        exercise = self.repo.get_by_id(question_id)
        if not exercise or not exercise.is_active:
            raise AppException("NOT_FOUND", f"题目{question_id}不存在", 404)
        
        # 2. 标准化答案
        normalized_answer = self._normalize_answer(exercise.type, answer)
        
        # 3. 检查幂等
        existing = self.repo.get_by_submission_id(user_id, client_submission_id)
        if existing:
            if (existing.question_id == question_id and 
                self._answers_equal(existing.answer, normalized_answer)):
                # 幂等返回
                return self._build_attempt_view(existing, exercise), False
            else:
                # 冲突
                raise AppException(
                    "CONFLICT",
                    "相同提交ID配了不同题目或答案",
                    409
                )
        
        # 4. 判分
        is_correct = self._judge(exercise, normalized_answer)
        
        # 5. 获取advisory lock（基于question_id哈希）
        lock_key = self._compute_lock_key(user_id, question_id)
        lock_acquired = self.db.execute(
            text(f"SELECT pg_try_advisory_xact_lock({lock_key})")
        ).scalar()
        
        if not lock_acquired:
            # 少数并发冲突，等待重试
            raise AppException("CONFLICT", "并发提交冲突，请重试", 409)
        
        # 6. 检查是否首次评估
        is_evaluation = not self.repo.has_evaluation(user_id, question_id)
        
        # 7. 插入
        attempt = ExerciseAttempt(
            user_id=user_id,
            question_id=question_id,
            client_submission_id=client_submission_id,
            answer=normalized_answer,
            is_correct=is_correct,
            is_evaluation=is_evaluation
        )
        self.db.add(attempt)
        self.db.commit()
        self.db.refresh(attempt)
        
        return self._build_attempt_view(attempt, exercise), True
    
    def get_attempts(
        self,
        user_id: UUID,
        question_id: Optional[str] = None,
        limit: int = 20,
        offset: int = 0
    ) -> tuple[List[AttemptView], int]:
        """查询本人答题历史"""
        attempts, total = self.repo.list_attempts(user_id, question_id, limit, offset)
        
        views = []
        for att in attempts:
            # 历史题目可能已下线，仍返回
            exercise = self.repo.get_by_id(att.question_id, include_inactive=True)
            if exercise:
                views.append(self._build_attempt_view(att, exercise))
        
        return views, total
    
    def _normalize_answer(self, q_type: str, answer: Any) -> Any:
        """标准化答案"""
        if q_type == "choice":
            if not isinstance(answer, str):
                raise AppException("VALIDATION_ERROR", "choice答案必须为字符串", 422)
            return answer.strip().upper()
        elif q_type == "judge":
            if not isinstance(answer, bool):
                raise AppException("VALIDATION_ERROR", "judge答案必须为boolean", 422)
            return answer
        else:
            raise AppException("VALIDATION_ERROR", f"不支持的题型{q_type}", 422)
    
    def _judge(self, exercise: Exercise, answer: Any) -> bool:
        """判分"""
        if exercise.type == "choice":
            return answer == exercise.answer_key
        elif exercise.type == "judge":
            return answer == exercise.answer_key
        return False
    
    def _answers_equal(self, a1: Any, a2: Any) -> bool:
        """比较答案相等（已标准化）"""
        return a1 == a2
    
    def _compute_lock_key(self, user_id: UUID, question_id: str) -> int:
        """计算advisory lock key"""
        import hashlib
        s = f"{user_id}:{question_id}"
        h = hashlib.sha256(s.encode()).digest()
        # 取前8字节转int，PostgreSQL advisory lock用bigint
        return int.from_bytes(h[:8], 'big', signed=True)
    
    def _build_attempt_view(self, attempt: ExerciseAttempt, exercise: Exercise) -> AttemptView:
        """构建AttemptView"""
        related = self.repo.get_related_nodes(exercise.question_id)
        
        student_view = ExerciseStudent(
            question_id=exercise.question_id,
            type=exercise.type,
            content=exercise.content,
            options=exercise.options,
            difficulty=exercise.difficulty,
            related_node_ids=[r.node_id for r in related],
            is_active=exercise.is_active
        )
        
        feedback = "本次回答正确" if attempt.is_correct else "本次回答有误，请复习关联知识节点"
        
        return AttemptView(
            attempt_id=attempt.attempt_id,
            client_submission_id=str(attempt.client_submission_id),
            question=student_view,
            answer=attempt.answer,
            is_correct=attempt.is_correct,
            is_evaluation=attempt.is_evaluation,
            submitted_at=attempt.submitted_at.isoformat(),
            feedback=feedback
        )
```

### backend/api/routers/exercise.py

```python
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from pydantic import BaseModel, Field
from uuid import UUID
from typing import Optional, Any
from common.database import get_db
from auth.service import require_password_changed, UserContext
from exercise.service import ExerciseService

router = APIRouter()

class AttemptRequest(BaseModel):
    question_id: str
    answer: Any  # choice为str，judge为bool
    client_submission_id: UUID

@router.get("/exercises")
def list_exercises(
    node_id: Optional[str] = None,
    question_id: Optional[str] = None,
    limit: int = Query(10, ge=1, le=20),
    offset: int = Query(0, ge=0),
    user: UserContext = Depends(require_password_changed),
    db: Session = Depends(get_db)
):
    service = ExerciseService(db)
    items, total = service.list_exercises(node_id, question_id, limit, offset)
    
    return {
        "data": {
            "items": [item.dict() for item in items],
            "total": total,
            "limit": limit,
            "offset": offset
        }
    }

@router.post("/exercises/attempts")
def submit_attempt(
    req: AttemptRequest,
    user: UserContext = Depends(require_password_changed),
    db: Session = Depends(get_db)
):
    service = ExerciseService(db)
    attempt_view, created = service.submit_attempt(
        user.user_id,
        req.question_id,
        req.answer,
        req.client_submission_id
    )
    
    status_code = 201 if created else 200
    return {
        "data": attempt_view.dict()
    }, status_code

@router.get("/exercises/attempts")
def get_attempts(
    question_id: Optional[str] = None,
    limit: int = Query(20, ge=1, le=100),
    offset: int = Query(0, ge=0),
    user: UserContext = Depends(require_password_changed),
    db: Session = Depends(get_db)
):
    service = ExerciseService(db)
    items, total = service.get_attempts(user.user_id, question_id, limit, offset)
    
    return {
        "data": {
            "items": [item.dict() for item in items],
            "total": total,
            "limit": limit,
            "offset": offset
        }
    }
```

### Cytoscape初始化（§5.2）

```typescript
// 必须在客户端useEffect或动态导入中初始化
import { useEffect, useRef } from 'react';

function GraphVisualization({ nodes, edges }) {
  const containerRef = useRef(null);
  
  useEffect(() => {
    if (!containerRef.current) return;
    
    // 动态导入避免SSR错误
    import('cytoscape').then((cytoscape) => {
      const cy = cytoscape.default({
        container: containerRef.current,
        elements: [
          ...nodes.map(n => ({ data: { id: n.node_id, label: n.title } })),
          ...edges.map(e => ({ 
            data: { 
              id: e.edge_id, 
              source: e.source_id, 
              target: e.target_id 
            } 
          }))
        ],
        style: [
          {
            selector: 'node',
            style: {
              'label': 'data(label)',
              'background-color': '#0070f3'
            }
          },
          {
            selector: 'edge',
            style: {
              'curve-style': 'bezier',
              'target-arrow-shape': 'triangle'
            }
          }
        ],
        layout: { name: 'cose' }
      });
      
      // 点击节点跳转
      cy.on('tap', 'node', (evt) => {
        const nodeId = evt.target.id();
        window.location.href = `/node/?id=${nodeId}`;
      });
    });
  }, [nodes, edges]);
  
  return <div ref={containerRef} style={{ width: '100%', height: '600px' }} />;
}
```

### web/app/login/page.tsx

```typescript
'use client';

import { useState, FormEvent } from 'react';
import { useRouter } from 'next/navigation';

export default function LoginPage() {
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const router = useRouter();
  
  const handleSubmit = async (e: FormEvent) => {
    e.preventDefault();
    setError('');
    
    try {
      const response = await fetch('/api/v1/auth/login', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'include',
        body: JSON.stringify({ username, password })
      });
      
      if (response.ok) {
        const data = await response.json();
        if (data.data.user.must_change_password) {
          router.push('/change-password/');
        } else {
          router.push('/graph/');
        }
      } else {
        const error = await response.json();
        setError(error.message || '登录失败');
      }
    } catch (err) {
      setError('网络错误');
    }
  };
  
  return (
    <div className="login-container">
      <h1>知识图谱学习系统</h1>
      <form onSubmit={handleSubmit}>
        <input
          type="text"
          placeholder="用户名"
          value={username}
          onChange={(e) => setUsername(e.target.value)}
          required
        />
        <input
          type="password"
          placeholder="密码"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          required
        />
        {error && <p className="error">{error}</p>}
        <button type="submit">登录</button>
      </form>
    </div>
  );
}
```

### web/lib/auth.ts

```typescript
export interface User {
  user_id: string;
  username: string;
  role: 'student' | 'teacher';
  must_change_password: boolean;
}

export async function getCurrentUser(): Promise<User | null> {
  try {
    const response = await fetch('/api/v1/auth/me', {
      credentials: 'include'
    });
    
    if (response.ok) {
      const data = await response.json();
      return data.data.user;
    }
    return null;
  } catch {
    return null;
  }
}

export async function logout(): Promise<void> {
  await fetch('/api/v1/auth/logout', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    credentials: 'include',
    body: '{}'
  });
  window.location.href = '/login/';
}
```

```typescript
'use client';

import { useEffect, useState, Suspense } from 'react';
import { useSearchParams } from 'next/navigation';
import GraphVisualization from '@/components/GraphVisualization';
import { getCurrentUser } from '@/lib/auth';

function GraphContent() {
  const searchParams = useSearchParams();
  const [user, setUser] = useState(null);
  const [subgraph, setSubgraph] = useState(null);
  
  useEffect(() => {
    getCurrentUser().then(u => {
      if (!u) window.location.href = '/login/';
      else if (u.must_change_password) window.location.href = '/change-password/';
      else setUser(u);
    });
  }, []);
  
  useEffect(() => {
    if (!user) return;
    
    const params = new URLSearchParams();
    if (searchParams.get('course_code')) params.set('course_code', searchParams.get('course_code'));
    if (searchParams.get('q')) params.set('q', searchParams.get('q'));
    if (searchParams.get('hops')) params.set('hops', searchParams.get('hops'));
    
    fetch(`/api/v1/graph/subgraph?${params}`, {
      credentials: 'include'
    })
      .then(r => r.json())
      .then(data => setSubgraph(data.data));
  }, [user, searchParams]);
  
  if (!user) return <div>加载中...</div>;
  
  return (
    <div>
      <h1>知识图谱</h1>
      <form>
        <select name="course_code" defaultValue={searchParams.get('course_code') || ''}>
          <option value="">全部课程</option>
          <option value="DS">数据结构</option>
          <option value="CO">计算机组成原理</option>
          <option value="OS">操作系统</option>
          <option value="CN">计算机网络</option>
        </select>
        <input name="q" placeholder="搜索节点" defaultValue={searchParams.get('q') || ''} />
        <button type="submit">查询</button>
      </form>
      
      {subgraph && (
        <GraphVisualization 
          nodes={subgraph.nodes} 
          edges={subgraph.edges} 
        />
      )}
    </div>
  );
}

export default function GraphPage() {
  return (
    <Suspense fallback={<div>加载中...</div>}>
      <GraphContent />
    </Suspense>
  );
}
```

```typescript
'use client';

import { useEffect, useState, Suspense } from 'react';
import { useSearchParams } from 'next/navigation';
import Link from 'next/link';

function NodeContent() {
  const searchParams = useSearchParams();
  const nodeId = searchParams.get('id');
  const [node, setNode] = useState(null);
  
  useEffect(() => {
    if (!nodeId) return;
    
    fetch(`/api/v1/nodes/${nodeId}`, { credentials: 'include' })
      .then(r => r.json())
      .then(data => {
        setNode(data.data);
        // 记录浏览事件
        fetch('/api/v1/learning/events', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          credentials: 'include',
          body: JSON.stringify({
            event_type: 'view_node',
            resource_type: 'node',
            resource_id: nodeId
          })
        });
      });
  }, [nodeId]);
  
  if (!node) return <div>加载中...</div>;
  
  return (
    <div>
      <h1>{node.node.title}</h1>
      <p>课程：{node.node.course_code}</p>
      <p>章节：{node.node.chapter_scope}</p>
      <p>描述：{node.node.description}</p>
      
      <h2>学习状态</h2>
      <p>状态：{node.learning.state}</p>
      <p>评估次数：{node.learning.evaluation_count}</p>
      {node.learning.recent_accuracy !== null && (
        <p>正确率：{(node.learning.recent_accuracy * 100).toFixed(0)}%</p>
      )}
      
      <h2>关联关系</h2>
      {node.node.outgoing_edges.map(e => (
        <div key={e.edge_id}>
          <Link href={`/edge/?id=${e.edge_id}`}>{e.display_label}</Link>
        </div>
      ))}
      
      <h2>练习题 ({node.exercise_count})</h2>
      <Link href={`/exercise/?node_id=${nodeId}`}>开始练习</Link>
    </div>
  );
}

export default function NodePage() {
  return (
    <Suspense>
      <NodeContent />
    </Suspense>
  );
}
```

### web/app/exercise/page.tsx

```typescript
'use client';

import { useEffect, useState, Suspense } from 'react';
import { useSearchParams } from 'next/navigation';
import { v4 as uuidv4 } from 'uuid';

function ExerciseContent() {
  const searchParams = useSearchParams();
  const questionId = searchParams.get('id');
  const [exercise, setExercise] = useState(null);
  const [answer, setAnswer] = useState(null);
  const [submissionId, setSubmissionId] = useState(uuidv4());
  const [result, setResult] = useState(null);
  
  useEffect(() => {
    if (!questionId) return;
    
    fetch(`/api/v1/exercises?question_id=${questionId}`, { credentials: 'include' })
      .then(r => r.json())
      .then(data => setExercise(data.data.items[0]));
  }, [questionId]);
  
  const handleSubmit = async () => {
    const response = await fetch('/api/v1/exercises/attempts', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      credentials: 'include',
      body: JSON.stringify({
        question_id: questionId,
        answer,
        client_submission_id: submissionId
      })
    });
    
    const data = await response.json();
    setResult(data.data);
    setSubmissionId(uuidv4());  // 成功后生成新UUID
  };
  
  if (!exercise) return <div>加载中...</div>;
  
  return (
    <div>
      <h2>{exercise.content}</h2>
      
      {exercise.type === 'choice' && (
        <div>
          {exercise.options.map(opt => (
            <label key={opt.key}>
              <input
                type="radio"
                name="answer"
                value={opt.key}
                onChange={(e) => setAnswer(e.target.value)}
              />
              {opt.key}. {opt.text}
            </label>
          ))}
        </div>
      )}
      
      {exercise.type === 'judge' && (
        <div>
          <label>
            <input
              type="radio"
              name="answer"
              value="true"
              onChange={() => setAnswer(true)}
            />
            正确
          </label>
          <label>
            <input
              type="radio"
              name="answer"
              value="false"
              onChange={() => setAnswer(false)}
            />
            错误
          </label>
        </div>
      )}
      
      <button onClick={handleSubmit} disabled={answer === null}>
        提交
      </button>
      
      {result && (
        <div className={result.is_correct ? 'correct' : 'wrong'}>
          {result.feedback}
        </div>
      )}
    </div>
  );
}

export default function ExercisePage() {
  return (
    <Suspense>
      <ExerciseContent />
    </Suspense>
  );
}
```

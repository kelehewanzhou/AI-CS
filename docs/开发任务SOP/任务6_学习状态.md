### 状态判定规则（§3.6）

```python
def compute_node_learning_state(
    user_id: UUID,
    node_id: str,
    db: Session
) -> NodeLearningState:
    """
    按§3.6优先级表计算学习状态
    
    N = 该节点相关题目的首次提交（is_evaluation=true）数量
    W = 按(submitted_at, attempt_id)取最近min(5, N)条首次提交
    R = W的正确率（正确数/条数）
    L = W中最新一条是否正确
    
    优先级1: N≥3 且 R≥80% 且 L=true → 已掌握
    优先级2: N>0，其他情况 → 练习中
    优先级3: N=0 且有view_node或start_exercise → 学习中
    优先级4: N=0 且无事件 → 未学习
    """
```

**R≥80%的整数判定**（避免浮点误差）：
```python
correct_count * 5 >= window_size * 4
```

```python
from typing import Optional
from uuid import UUID
from sqlalchemy.orm import Session
from sqlalchemy import text

class LearningRules:
    """学习状态规则，只属于learning模块"""
    
    RULE_VERSION = "p0-first-distinct-v1"
    
    @staticmethod
    def compute_state(
        user_id: UUID,
        node_id: str,
        db: Session
    ) -> dict:
        """计算单个节点的学习状态"""
        
        # 1. 查询有效关联题目数
        result = db.execute(text("""
            SELECT COUNT(DISTINCT e.question_id)
            FROM exercises e
            JOIN exercise_related_nodes ern ON e.question_id = ern.question_id
            WHERE ern.node_id = :node_id AND e.is_active = true
        """), {"node_id": node_id})
        available_count = result.scalar() or 0
        can_evaluate = available_count >= 3
        
        # 2. 查询首次提交（评估）
        result = db.execute(text("""
            SELECT 
                ea.is_correct,
                ea.submitted_at,
                ea.attempt_id
            FROM exercise_attempts ea
            JOIN exercise_related_nodes ern ON ea.question_id = ern.question_id
            WHERE ea.user_id = :user_id
              AND ern.node_id = :node_id
              AND ea.is_evaluation = true
            ORDER BY ea.submitted_at DESC, ea.attempt_id DESC
        """), {"user_id": user_id, "node_id": node_id})
        evaluations = result.fetchall()
        
        N = len(evaluations)
        
        # 3. 查询总答题数（含重做）
        result = db.execute(text("""
            SELECT COUNT(*)
            FROM exercise_attempts ea
            JOIN exercise_related_nodes ern ON ea.question_id = ern.question_id
            WHERE ea.user_id = :user_id AND ern.node_id = :node_id
        """), {"user_id": user_id, "node_id": node_id})
        total_attempts = result.scalar() or 0
        
        # 4. 查询view_node或start_exercise事件
        result = db.execute(text("""
            SELECT EXISTS (
                SELECT 1 FROM learning_events
                WHERE user_id = :user_id
                  AND (
                    (event_type = 'view_node' AND node_id = :node_id) OR
                    (event_type = 'start_exercise' AND question_id IN (
                        SELECT question_id FROM exercise_related_nodes WHERE node_id = :node_id
                    ))
                  )
            )
        """), {"user_id": user_id, "node_id": node_id})
        has_viewed = result.scalar()
        
        # 5. 计算状态
        if N == 0:
            if has_viewed:
                if available_count == 0:
                    state, reason = "学习中", "no_exercises"
                else:
                    state, reason = "学习中", "viewed_or_started"
            else:
                state, reason = "未学习", "no_activity"
            
            return {
                "node_id": node_id,
                "state": state,
                "reason": reason,
                "has_viewed": has_viewed,
                "total_attempt_count": total_attempts,
                "evaluation_count": N,
                "recent_window_size": 0,
                "recent_accuracy": None,
                "latest_evaluation_correct": None,
                "available_exercise_count": available_count,
                "can_evaluate_mastery": can_evaluate,
                "rule_version": LearningRules.RULE_VERSION
            }
        
        # N > 0
        window_size = min(5, N)
        recent_evals = evaluations[:window_size]
        correct_count = sum(1 for e in recent_evals if e.is_correct)
        latest_correct = recent_evals[0].is_correct
        
        # 整数判定：correct_count * 5 >= window_size * 4
        accuracy_meets = (correct_count * 5 >= window_size * 4)
        
        if N >= 3 and accuracy_meets and latest_correct:
            state, reason = "已掌握", "mastery_rule_met"
        else:
            # 练习中，细分原因
            if N < 3:
                reason = "insufficient_distinct_questions"
            elif not latest_correct:
                reason = "latest_evaluation_wrong"
            else:
                reason = "accuracy_below_threshold"
            state = "练习中"
        
        return {
            "node_id": node_id,
            "state": state,
            "reason": reason,
            "has_viewed": has_viewed,
            "total_attempt_count": total_attempts,
            "evaluation_count": N,
            "recent_window_size": window_size,
            "recent_accuracy": correct_count / window_size,
            "latest_evaluation_correct": latest_correct,
            "available_exercise_count": available_count,
            "can_evaluate_mastery": can_evaluate,
            "rule_version": LearningRules.RULE_VERSION
        }
```

### backend/learning/service.py

```python
from typing import List, Optional
from uuid import UUID
from sqlalchemy.orm import Session
from common.errors import AppException
from .repository import LearningRepository
from .rules import LearningRules
from graph.repository import GraphRepository

class LearningService:
    def __init__(self, db: Session):
        self.db = db
        self.repo = LearningRepository(db)
        self.graph_repo = GraphRepository(db)
    
    def record_event(
        self,
        user_id: UUID,
        event_type: str,
        resource_id: str
    ) -> dict:
        """记录学习事件"""
        # 验证事件类型
        if event_type not in ["view_node", "view_edge", "start_exercise"]:
            raise AppException("VALIDATION_ERROR", f"非法事件类型{event_type}", 422)
        
        # 验证资源存在
        if event_type == "view_node":
            node = self.graph_repo.get_node_by_id(resource_id)
            if not node or not node.is_active:
                raise AppException("NOT_FOUND", f"节点{resource_id}不存在", 404)
            event = self.repo.create_event(user_id, event_type, node_id=resource_id)
        
        elif event_type == "view_edge":
            edge = self.graph_repo.get_edge_by_id(resource_id)
            if not edge or not edge.is_active:
                raise AppException("NOT_FOUND", f"关系{resource_id}不存在", 404)
            event = self.repo.create_event(user_id, event_type, edge_id=resource_id)
        
        else:  # start_exercise
            # 验证题目（由exercise模块提供）
            from exercise.repository import ExerciseRepository
            ex_repo = ExerciseRepository(self.db)
            exercise = ex_repo.get_by_id(resource_id)
            if not exercise or not exercise.is_active:
                raise AppException("NOT_FOUND", f"题目{resource_id}不存在", 404)
            event = self.repo.create_event(user_id, event_type, question_id=resource_id)
        
        self.db.commit()
        
        return {
            "event_id": event.event_id,
            "event_type": event.event_type,
            "resource_type": event_type.replace("view_", "").replace("start_", ""),
            "resource_id": resource_id,
            "created_at": event.created_at.isoformat()
        }
    
    def get_events(
        self,
        user_id: UUID,
        event_type: Optional[str] = None,
        limit: int = 20,
        cursor: Optional[str] = None
    ) -> tuple[List[dict], Optional[str]]:
        """查询事件历史，游标分页"""
        # 解析游标
        if cursor:
            cursor_data = self._verify_cursor(cursor, user_id, "events", event_type)
            after_created_at = cursor_data["created_at"]
            after_event_id = cursor_data["event_id"]
        else:
            after_created_at = None
            after_event_id = None
        
        events = self.repo.list_events(
            user_id, event_type, limit + 1,
            after_created_at, after_event_id
        )
        
        has_more = len(events) > limit
        if has_more:
            events = events[:limit]
        
        items = []
        for e in events:
            resource_type = e.event_type.replace("view_", "").replace("start_", "")
            resource_id = e.node_id or e.edge_id or e.question_id
            items.append({
                "event_id": e.event_id,
                "event_type": e.event_type,
                "resource_type": resource_type,
                "resource_id": resource_id,
                "created_at": e.created_at.isoformat()
            })
        
        next_cursor = None
        if has_more:
            last = events[-1]
            next_cursor = self._generate_cursor(
                user_id, "events", event_type,
                last.created_at, last.event_id
            )
        
        return items, next_cursor
    
    def get_states(
        self,
        user_id: UUID,
        node_ids: Optional[List[str]] = None
    ) -> List[dict]:
        """查询节点学习状态"""
        if node_ids:
            # 验证节点存在
            for nid in node_ids:
                node = self.graph_repo.get_node_by_id(nid)
                if not node or not node.is_active:
                    raise AppException("NOT_FOUND", f"节点{nid}不存在", 404)
            target_nodes = node_ids
        else:
            # 全部有效节点
            all_nodes = self.graph_repo.get_all_active_nodes()
            target_nodes = [n.node_id for n in all_nodes]
        
        states = []
        for node_id in target_nodes:
            state = LearningRules.compute_state(user_id, node_id, self.db)
            states.append(state)
        
        return states
    
    def get_content_summary(self) -> dict:
        """教师内容验收摘要"""
        # 最近成功导入
        result = self.db.execute(text("""
            SELECT run_id, checksum, finished_at, counts
            FROM import_runs
            WHERE status = 'succeeded'
            ORDER BY finished_at DESC
            LIMIT 1
        """))
        last_import = result.fetchone()
        
        if not last_import:
            raise AppException("UNAVAILABLE", "未导入种子数据", 503)
        
        counts_data = last_import.counts
        
        # 证据分布
        result = self.db.execute(text("""
            SELECT 
                evidence_status,
                COUNT(*) as cnt
            FROM knowledge_edges
            WHERE is_active = true
            GROUP BY evidence_status
        """))
        evidence_rows = result.fetchall()
        evidence_counts = {
            "verified": 0,
            "partial": 0,
            "pending": 0
        }
        for row in evidence_rows:
            if row.evidence_status == "已核验":
                evidence_counts["verified"] = row.cnt
            elif row.evidence_status == "部分定位":
                evidence_counts["partial"] = row.cnt
            else:
                evidence_counts["pending"] = row.cnt
        
        return {
            "content_scope": "current_seed",
            "seed_checksum": last_import.checksum,
            "last_import": {
                "run_id": last_import.run_id,
                "finished_at": last_import.finished_at.isoformat()
            },
            "counts": counts_data,
            "edge_evidence_counts": evidence_counts
        }
    
    def _generate_cursor(self, user_id, scope, filter_val, created_at, record_id):
        """生成签名游标"""
        import hmac
        import hashlib
        import base64
        import json
        
        payload = {
            "user_id": str(user_id),
            "scope": scope,
            "filter": filter_val,
            "created_at": created_at.isoformat(),
            "record_id": record_id
        }
        data = json.dumps(payload, sort_keys=True)
        signature = hmac.new(
            settings.session_secret_key.encode(),
            data.encode(),
            hashlib.sha256
        ).hexdigest()
        
        cursor_obj = {"data": payload, "sig": signature}
        return base64.urlsafe_b64encode(
            json.dumps(cursor_obj).encode()
        ).decode()
    
    def _verify_cursor(self, cursor, expected_user_id, scope, filter_val):
        """验证游标"""
        import hmac
        import hashlib
        import base64
        import json
        
        try:
            cursor_obj = json.loads(base64.urlsafe_b64decode(cursor))
            payload = cursor_obj["data"]
            sig = cursor_obj["sig"]
            
            # 验证签名
            data = json.dumps(payload, sort_keys=True)
            expected_sig = hmac.new(
                settings.session_secret_key.encode(),
                data.encode(),
                hashlib.sha256
            ).hexdigest()
            
            if not hmac.compare_digest(sig, expected_sig):
                raise ValueError("签名不匹配")
            
            # 验证绑定
            if payload["user_id"] != str(expected_user_id):
                raise ValueError("用户不匹配")
            if payload["scope"] != scope:
                raise ValueError("scope不匹配")
            if payload.get("filter") != filter_val:
                raise ValueError("过滤条件不匹配")
            
            return payload
        except Exception:
            raise AppException("VALIDATION_ERROR", "非法游标", 422)
```

```python
@router.get("/nodes/{node_id}")
def get_node_with_learning(
    node_id: str,
    user: UserContext = Depends(require_password_changed),
    db: Session = Depends(get_db)
):
    """节点详情，组合graph+exercise+learning"""
    # 开启只读事务，确保一致性快照
    db.execute(text("SET TRANSACTION ISOLATION LEVEL REPEATABLE READ READ ONLY"))
    
    graph_service = GraphService(db)
    node = graph_service.get_node(node_id)
    
    # exercise_count
    from exercise.repository import ExerciseRepository
    ex_repo = ExerciseRepository(db)
    ex_count = ex_repo.count_by_node(node_id)
    
    # learning状态
    learning_service = LearningService(db)
    states = learning_service.get_states(user.user_id, [node_id])
    
    return {
        "data": {
            "node": node.dict(),
            "exercise_count": ex_count,
            "learning": states[0] if states else {}
        }
    }
```

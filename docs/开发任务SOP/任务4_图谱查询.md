### 节点证据状态计算（§2.4）

```python
def compute_node_evidence_status(edges: List[Edge]) -> dict:
    """
    根据关联边计算节点证据状态
    - 存在待补充 → 待补充
    - 否则存在部分定位 → 部分定位
    - 否则有边 → 已核验
    - 无边 → 待补充，reason=no_relations
    """
    if not edges:
        return {
            "status": "待补充",
            "reason": "no_relations",
            "counts": {"verified": 0, "partial": 0, "pending": 0},
            "edge_ids": []
        }
    
    counts = {"verified": 0, "partial": 0, "pending": 0}
    edge_ids = []
    
    for edge in edges:
        edge_ids.append(edge.edge_id)
        if edge.evidence_status == "已核验":
            counts["verified"] += 1
        elif edge.evidence_status == "部分定位":
            counts["partial"] += 1
        else:
            counts["pending"] += 1
    
    # 按最弱规则
    if counts["pending"] > 0:
        status = "待补充"
    elif counts["partial"] > 0:
        status = "部分定位"
    else:
        status = "已核验"
    
    return {
        "status": status,
        "reason": "incident_edges",
        "counts": counts,
        "edge_ids": edge_ids
    }
```

### 子图遍历语义（§3.3）

1. **有node_id**：起点为该节点
2. **无node_id但有course/q**：所有符合条件的有效节点为起点
3. **无任何条件**：overview模式，返回全部有效节点及过滤后的边，applied_hops=0
4. **有起点**：沿过滤后的入边和出边BFS扩展，最多2层
5. **对称关系**：按source_id < target_id存储，遍历时双向可达
6. **上限**：60节点、120边，超限返回422

### backend/graph/service.py

```python
from typing import List, Optional
from sqlalchemy.orm import Session
from common.errors import AppException
from .repository import GraphRepository
from .schemas import SubgraphQuery, SubgraphResult, NodeContent, EdgeView

class GraphService:
    def __init__(self, db: Session):
        self.db = db
        self.repo = GraphRepository(db)
    
    def get_subgraph(self, query: SubgraphQuery, seed_checksum: str) -> SubgraphResult:
        """查询子图"""
        # 1. 确定起点
        if query.node_id:
            # 中心模式
            seed_node = self.repo.get_node_by_id(query.node_id)
            if not seed_node or not seed_node.is_active:
                raise AppException("NOT_FOUND", f"节点{query.node_id}不存在", 404)
            if query.course_code and seed_node.course_code != query.course_code:
                raise AppException("VALIDATION_ERROR", "中心节点不属于指定课程", 422)
            seed_nodes = [seed_node]
            mode = "centered"
        elif query.course_code or query.q:
            # 多起点模式
            seed_nodes = self.repo.find_nodes(query.course_code, query.q)
            if not seed_nodes:
                return self._empty_subgraph(query, seed_checksum)
            mode = "filtered"
        else:
            # Overview模式
            all_nodes = self.repo.get_all_active_nodes()
            filtered_edges = self.repo.get_filtered_edges(
                query.relation_type, 
                query.evidence_status
            )
            return self._build_overview(all_nodes, filtered_edges, query, seed_checksum)
        
        # 2. 过滤边
        filtered_edges = self.repo.get_filtered_edges(
            query.relation_type,
            query.evidence_status
        )
        
        # 3. BFS遍历
        visited_nodes = {n.node_id: n for n in seed_nodes}
        current_layer = [n.node_id for n in seed_nodes]
        
        for hop in range(query.hops):
            next_layer = set()
            for node_id in current_layer:
                # 入边和出边（对称关系双向可达）
                neighbors = self.repo.get_neighbors(node_id, filtered_edges)
                for neighbor_id in neighbors:
                    if neighbor_id not in visited_nodes:
                        neighbor = self.repo.get_node_by_id(neighbor_id)
                        if neighbor and neighbor.is_active:
                            visited_nodes[neighbor_id] = neighbor
                            next_layer.add(neighbor_id)
            current_layer = list(next_layer)
        
        # 4. 诱导子图：只保留visited_nodes之间的边
        result_edges = [
            e for e in filtered_edges
            if e.source_id in visited_nodes and e.target_id in visited_nodes
        ]
        
        # 5. 检查上限
        if len(visited_nodes) > 60 or len(result_edges) > 120:
            raise AppException(
                "VALIDATION_ERROR",
                "子图超过限制（60节点/120边），请缩小查询范围",
                422,
                {"reason": "subgraph_limit"}
            )
        
        # 6. 构建响应
        return self._build_subgraph_result(
            list(visited_nodes.values()),
            result_edges,
            mode,
            [n.node_id for n in seed_nodes],
            query,
            seed_checksum
        )
    
    def get_node(self, node_id: str) -> NodeContent:
        """获取节点详情"""
        node = self.repo.get_node_by_id(node_id)
        if not node or not node.is_active:
            raise AppException("NOT_FOUND", f"节点{node_id}不存在", 404)
        
        # 关联边
        edges = self.repo.get_incident_edges(node_id)
        
        # 证据概览
        evidence_overview = self._compute_evidence_overview(edges)
        
        # 分离入边和出边
        incoming = [e for e in edges if e.target_id == node_id]
        outgoing = [e for e in edges if e.source_id == node_id]
        
        return NodeContent(
            node_id=node.node_id,
            title=node.title,
            course_code=node.course_code,
            chapter_scope=node.chapter_scope,
            description=node.description,
            aliases=node.aliases,
            scope_tags=node.scope_tags,
            evidence_overview=evidence_overview,
            incoming_edges=[self._build_edge_view(e) for e in incoming],
            outgoing_edges=[self._build_edge_view(e) for e in outgoing]
        )
    
    def get_edge(self, edge_id: str) -> EdgeView:
        """获取关系详情"""
        edge = self.repo.get_edge_by_id(edge_id)
        if not edge or not edge.is_active:
            raise AppException("NOT_FOUND", f"关系{edge_id}不存在", 404)
        
        # 检查端点是否有效
        source = self.repo.get_node_by_id(edge.source_id)
        target = self.repo.get_node_by_id(edge.target_id)
        if not source or not source.is_active or not target or not target.is_active:
            raise AppException("NOT_FOUND", "关系端点已下线", 404)
        
        return self._build_edge_view(edge)
```

### backend/api/routers/graph.py

```python
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from typing import List, Optional
from common.database import get_db
from auth.service import require_password_changed
from graph.service import GraphService
from graph.schemas import SubgraphQuery

router = APIRouter()

@router.get("/subgraph")
def get_subgraph(
    node_id: Optional[str] = None,
    course_code: Optional[str] = Query(None, pattern="^(DS|CO|OS|CN)$"),
    q: Optional[str] = Query(None, min_length=1, max_length=100),
    hops: int = Query(1, ge=1, le=2),
    relation_type: List[str] = Query([]),
    evidence_status: List[str] = Query([]),
    user = Depends(require_password_changed),
    db: Session = Depends(get_db)
):
    # 获取当前种子checksum
    from sqlalchemy import text
    result = db.execute(text(
        "SELECT checksum FROM import_runs WHERE status='succeeded' ORDER BY finished_at DESC LIMIT 1"
    ))
    seed_checksum = result.scalar() or ""
    
    query = SubgraphQuery(
        node_id=node_id,
        course_code=course_code,
        q=q,
        hops=hops,
        relation_type=relation_type,
        evidence_status=evidence_status
    )
    
    graph_service = GraphService(db)
    result = graph_service.get_subgraph(query, seed_checksum)
    
    return {"data": result.dict()}

@router.get("/nodes/{node_id}")
def get_node(
    node_id: str,
    user = Depends(require_password_changed),
    db: Session = Depends(get_db)
):
    graph_service = GraphService(db)
    node = graph_service.get_node(node_id)
    
    # exercise_count由API层组合（任务5后添加）
    return {
        "data": {
            "node": node.dict(),
            "exercise_count": 0,  # 任务5填充
            "learning": {}  # 任务6填充
        }
    }

@router.get("/edges/{edge_id}")
def get_edge(
    edge_id: str,
    user = Depends(require_password_changed),
    db: Session = Depends(get_db)
):
    graph_service = GraphService(db)
    edge = graph_service.get_edge(edge_id)
    
    return {"data": edge.dict()}
```

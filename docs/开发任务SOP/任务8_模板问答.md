### 归一化规则（§3.7）

```python
import unicodedata
import re

def normalize_text(text: str) -> str:
    """
    归一化文本用于匹配
    1. Unicode NFKC规范化
    2. casefold大小写折叠
    3. 标点空白替换为空格并折叠
    4. trim
    """
    # NFKC: 全角→半角、兼容字符→标准字符
    text = unicodedata.normalize('NFKC', text)
    
    # casefold: 大小写不敏感
    text = text.casefold()
    
    # 标点和空白替换为空格
    text = re.sub(r'[^\w\s]', ' ', text, flags=re.UNICODE)
    text = re.sub(r'\s+', ' ', text)
    
    return text.strip()
```

### 关系关键词字典（§3.7）

```python
RELATION_KEYWORDS = {
    'prerequisite': ['前置', '先学', '基础'],
    'implements': ['实现', '支撑', '实施'],
    'contrast': ['比较', '对比', '区别', '差异'],
    'boundary': ['边界', '职责', '划分']
}
```

### 匹配规则（§3.7）

1. **标题/别名匹配**：
   - 标准化后做子串匹配
   - ASCII纯字母数字别名使用单词边界（避免"OS"命中"most"）
   - 跨度有包含关系时保留更长词
   - 同一节点多次命中只算一次

2. **排序**：
   - 标题命中优先于别名命中
   - 最长命中长度降序
   - node_id升序

3. **节点数限制**：
   - 最多处理2个节点
   - 超过2个返回not_covered/too_many_nodes

### 响应结构（§3.7）

```python
class QAResponse(BaseModel):
```

### backend/qa/service.py

```python
import unicodedata
import re
from typing import List, Tuple, Optional
from uuid import UUID
from sqlalchemy.orm import Session
from common.errors import AppException
from graph.service import GraphService

class QAService:
    def __init__(self, db: Session):
        self.db = db
        self.graph_service = GraphService(db)
    
    RELATION_KEYWORDS = {
        'prerequisite': ['前置', '先学', '基础'],
        'implements': ['实现', '支撑', '实施'],
        'contrast': ['比较', '对比', '区别', '差异'],
        'boundary': ['边界', '职责', '划分']
    }
    
    def ask(self, user_id: UUID, question: str, request_id: str) -> dict:
        """
        问答主流程
        """
        try:
            # 1. 归一化问题
            normalized_q = self._normalize_text(question.strip())
            if not normalized_q or len(normalized_q) < 2:
                raise AppException("VALIDATION_ERROR", "问题过短", 422)
            
            # 2. 获取当前节点字典
            nodes_dict = self._load_qa_dictionary()
            
            # 3. 匹配节点
            matched_nodes = self._match_nodes(normalized_q, nodes_dict)
            
            # 4. 根据匹配数量选择模式
            if len(matched_nodes) == 0:
                return self._build_response(
                    "not_covered", "no_match",
                    "当前知识图谱未覆盖这个问题，请尝试更具体的知识点名称。",
                    [], [], True, "",
                    user_id, question, request_id
                )
            
            elif len(matched_nodes) > 2:
                return self._build_response(
                    "not_covered", "too_many_nodes",
                    f"问题涉及{len(matched_nodes)}个知识点，请拆分为单个或两个知识点的问题。",
                    [], [], True, "",
                    user_id, question, request_id
                )
            
            elif len(matched_nodes) == 1:
                # 单节点概念模式
                return self._answer_concept(matched_nodes[0], user_id, question, request_id)
            
            else:  # len == 2
                # 双节点关系模式
                return self._answer_relation(
                    matched_nodes[0], matched_nodes[1],
                    normalized_q, user_id, question, request_id
                )
        
        except AppException:
            raise
        except Exception as e:
            # 异常隔离：QA失败不影响图谱查询
            import logging
            logging.error(f"QA异常 request_id={request_id}: {e}")
            raise AppException("INTERNAL_ERROR", "问答服务暂时不可用", 500)
    
    def _normalize_text(self, text: str) -> str:
        """归一化文本"""
        text = unicodedata.normalize('NFKC', text)
        text = text.casefold()
        text = re.sub(r'[^\w\s]', ' ', text, flags=re.UNICODE)
        text = re.sub(r'\s+', ' ', text)
        return text.strip()
    
    def _load_qa_dictionary(self) -> List[dict]:
        """从数据库加载当前有效节点及别名"""
        from graph.repository import GraphRepository
        repo = GraphRepository(self.db)
        nodes = repo.get_all_active_nodes()
        
        result = []
        for node in nodes:
            result.append({
                'node_id': node.node_id,
                'title': node.title,
                'normalized_title': self._normalize_text(node.title),
                'aliases': [
                    {'original': alias, 'normalized': self._normalize_text(alias)}
                    for alias in node.aliases
                    if len(self._normalize_text(alias)) >= 2
                ],
                'description': node.description
            })
        return result
    
    def _match_nodes(self, normalized_q: str, nodes_dict: List[dict]) -> List[dict]:
        """匹配节点，返回排序后的匹配列表"""
        matches = []
        
        for node_dict in nodes_dict:
            # 标题匹配
            title_norm = node_dict['normalized_title']
            if title_norm in normalized_q:
                matches.append({
                    'node': node_dict,
                    'match_type': 'title',
                    'match_length': len(title_norm)
                })
                continue  # 同一节点只算一次
            
            # 别名匹配
            for alias_info in node_dict['aliases']:
                alias_norm = alias_info['normalized']
                
                # ASCII纯字母数字：使用单词边界
                if re.match(r'^[a-z0-9]+$', alias_norm):
                    pattern = r'\b' + re.escape(alias_norm) + r'\b'
                    if re.search(pattern, normalized_q):
                        matches.append({
                            'node': node_dict,
                            'match_type': 'alias',
                            'match_length': len(alias_norm)
                        })
                        break
                else:
                    # 非ASCII：子串匹配
                    if alias_norm in normalized_q:
                        matches.append({
                            'node': node_dict,
                            'match_type': 'alias',
                            'match_length': len(alias_norm)
                        })
                        break
        
        # 去重跨度包含：保留更长匹配
        # （简化实现，完整版需要位置信息）
        
        # 排序：标题优先、长度降序、node_id升序
        matches.sort(key=lambda m: (
            0 if m['match_type'] == 'title' else 1,
            -m['match_length'],
            m['node']['node_id']
        ))
        
        return [m['node'] for m in matches]
    
    def _answer_concept(self, node_dict: dict, user_id: UUID, question: str, request_id: str) -> dict:
        """单节点概念模式"""
        node_id = node_dict['node_id']
        
        # 获取关联边，筛选已核验
        edges = self.graph_service.repo.get_incident_edges(node_id)
        verified_edges = [e for e in edges if e.is_active and e.evidence_status == '已核验']
        
        if not verified_edges:
            return self._build_response(
                "insufficient_evidence", None,
                f"知识点"{node_dict['title']}"暂无已核验的关系证据，无法提供可靠解释。",
                [node_id], [], True,
                "缺少已核验关系",
                user_id, question, request_id
            )
        
        # 最多3条边
        selected_edges = sorted(verified_edges, key=lambda e: e.edge_id)[:3]
        
        # 构建答案
        answer_parts = [
            f"{node_dict['title']}：{node_dict['description']}"
        ]
        
        for edge in selected_edges:
            if edge.source_id == node_id:
                other_id = edge.target_id
                direction = "支撑"
            else:
                other_id = edge.source_id
                direction = "依赖于"
            
            other_node = self.graph_service.repo.get_node_by_id(other_id)
            if other_node:
                answer_parts.append(
                    f"- {direction}{other_node.title}：{edge.mechanism}"
                )
        
        answer_parts.append("\n（以上内容基于课程教材关系，非教材原文）")
        
        # 收集使用的节点和边ID
        used_node_ids = {node_id}
        used_edge_ids = []
        boundary_notes = []
        
        for edge in selected_edges:
            used_node_ids.add(edge.source_id)
            used_node_ids.add(edge.target_id)
            used_edge_ids.append(edge.edge_id)
            if edge.boundary_note:
                boundary_notes.append(edge.boundary_note)
        
        return self._build_response(
            "answered", None,
            "\n".join(answer_parts),
            sorted(used_node_ids), sorted(used_edge_ids),
            False, " | ".join(boundary_notes),
            user_id, question, request_id
        )
    
    def _answer_relation(
        self, node1: dict, node2: dict,
        normalized_q: str, user_id: UUID, question: str, request_id: str
    ) -> dict:
        """双节点关系模式"""
        node1_id = node1['node_id']
        node2_id = node2['node_id']
        
        # 检测关系类型关键词
        detected_types = []
        for rel_type, keywords in self.RELATION_KEYWORDS.items():
            if any(kw in normalized_q for kw in keywords):
                detected_types.append(rel_type)
        
        if len(detected_types) > 1:
            return self._build_response(
                "not_covered", "ambiguous_relation",
                f"问题同时涉及多种关系类型，请一次只询问一种关系（如前置、实现、对比或边界）。",
                [node1_id, node2_id], [], True, "",
                user_id, question, request_id
            )
        
        target_type = detected_types[0] if detected_types else None
        
        # 查询直接关系（双向）
        direct_edges = self._find_direct_edges(node1_id, node2_id, target_type)
        
        if not direct_edges:
            return self._build_response(
                "not_covered", "no_direct_relation",
                f"{node1['title']}与{node2['title']}之间没有记录的直接关系。",
                [node1_id, node2_id], [], True,
                "未记录直接关系，不推断多跳关系",
                user_id, question, request_id
            )
        
        # 筛选已核验边
        verified = [e for e in direct_edges if e.evidence_status == '已核验']
        if not verified:
            return self._build_response(
                "insufficient_evidence", None,
                f"{node1['title']}与{node2['title']}的关系尚未完成教材核验。",
                [node1_id, node2_id], [e.edge_id for e in direct_edges],
                True, "证据待核验",
                user_id, question, request_id
            )
        
        # 最多3条
        selected = sorted(verified, key=lambda e: (e.relation_type, e.edge_id))[:3]
        
        # 生成答案
        answer_parts = [f"{node1['title']}与{node2['title']}的关系：\n"]
        used_node_ids = {node1_id, node2_id}
        used_edge_ids = []
        boundary_notes = []
        
        for edge in selected:
            # 使用存储方向的display_label
            label = self.graph_service._generate_display_label(edge)
            answer_parts.append(f"- {label}：{edge.mechanism}")
            used_edge_ids.append(edge.edge_id)
            if edge.boundary_note:
                boundary_notes.append(edge.boundary_note)
        
        answer_parts.append("\n（以上基于课程教材关系）")
        
        return self._build_response(
            "answered", None,
            "\n".join(answer_parts),
            sorted(used_node_ids), sorted(used_edge_ids),
            False, " | ".join(boundary_notes),
            user_id, question, request_id
        )
    
    def _find_direct_edges(self, node1_id: str, node2_id: str, target_type: Optional[str]) -> List:
        """查找两节点间的直接边"""
        from sqlalchemy import text
        
        query = """
            SELECT * FROM knowledge_edges
            WHERE is_active = true
              AND (
                (source_id = :n1 AND target_id = :n2) OR
                (source_id = :n2 AND target_id = :n1)
              )
        """
        params = {'n1': node1_id, 'n2': node2_id}
        
        if target_type:
            query += " AND relation_type = :rel_type"
            params['rel_type'] = target_type
        
        result = self.db.execute(text(query), params)
        edges = result.fetchall()
        
        # 转换为模型对象
        from graph.models import KnowledgeEdge
        return [KnowledgeEdge(**dict(row._mapping)) for row in edges]
    
    def _build_response(
        self, status: str, reason: Optional[str], answer: str,
        used_node_ids: List[str], used_edge_ids: List[str],
        uncertainty_flag: bool, boundary_note: str,
        user_id: UUID, question: str, request_id: str
    ) -> dict:
        """构建响应并写入qa_records"""
        import hashlib
        import hmac
        from datetime import datetime
        from common.config import settings
        
        # actor_key: HMAC(user_id, 月度密钥)
        month_key = datetime.now().strftime("%Y%m")
        actor_key = hmac.new(
            f"{settings.session_secret_key}:{month_key}".encode(),
            str(user_id).encode(),
            hashlib.sha256
        ).hexdigest()
        
        # 写入记录
        from sqlalchemy import text
        self.db.execute(text("""
            INSERT INTO qa_records (
                question_text, answer_text, status, reason,
                used_node_ids, used_edge_ids, uncertainty_flag, boundary_note,
                actor_key, request_id
            ) VALUES (
                :question, :answer, :status, :reason,
                :node_ids, :edge_ids, :uncertainty, :boundary,
                :actor_key, :request_id
            )
        """), {
            'question': question[:500],
            'answer': answer,
            'status': status,
            'reason': reason or '',
            'node_ids': used_node_ids,
            'edge_ids': used_edge_ids,
            'uncertainty': uncertainty_flag,
            'boundary': boundary_note,
            'actor_key': actor_key,
            'request_id': request_id
        })
        self.db.commit()
        
        return {
            'status': status,
            'reason': reason,
            'answer': answer,
            'used_node_ids': used_node_ids,
            'used_edge_ids': used_edge_ids,
            'uncertainty_flag': uncertainty_flag,
            'boundary_note': boundary_note
        }
```

### backend/api/routers/qa.py

```python
from fastapi import APIRouter, Depends, Request
from sqlalchemy.orm import Session
from pydantic import BaseModel, Field
from common.database import get_db
from common.errors import get_request_id
from auth.service import require_password_changed, UserContext
from qa.service import QAService

router = APIRouter()

class AskRequest(BaseModel):
    question: str = Field(min_length=1, max_length=500)

@router.post("/ask")
def ask_question(
    req: AskRequest,
    request: Request,
    user: UserContext = Depends(require_password_changed),
    db: Session = Depends(get_db)
):
    request_id = get_request_id(request)
    qa_service = QAService(db)
    
    result = qa_service.ask(user.user_id, req.question, request_id)
    
    return {"data": result}
```

```typescript
'use client';

import { useState } from 'react';

export default function AskPanel() {
  const [question, setQuestion] = useState('');
  const [result, setResult] = useState<any>(null);
  const [loading, setLoading] = useState(false);
  
  const handleAsk = async () => {
    if (!question.trim()) return;
    
    setLoading(true);
    try {
      const response = await fetch('/api/v1/ask', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'include',
        body: JSON.stringify({ question })
      });
      
      const data = await response.json();
      setResult(data.data);
    } catch (err) {
      setResult({ status: 'error', answer: '问答服务暂时不可用' });
    } finally {
      setLoading(false);
    }
  };
  
  return (
    <div className="ask-panel">
      <h3>知识问答</h3>
      <input
        type="text"
        value={question}
        onChange={(e) => setQuestion(e.target.value)}
        placeholder="输入知识点名称，如：什么是TLB？"
        maxLength={500}
      />
      <button onClick={handleAsk} disabled={loading}>
        {loading ? '查询中...' : '提问'}
      </button>
      
      {result && (
        <div className={`answer ${result.status}`}>
          {result.status === 'answered' && (
            <>
              <div className="answer-text">{result.answer}</div>
              {result.used_node_ids.length > 0 && (
                <div className="references">
                  引用节点：{result.used_node_ids.join(', ')}
                </div>
              )}
            </>
          )}
          
          {result.status === 'not_covered' && (
            <div className="not-covered">
              <p>{result.answer}</p>
              <small>原因：{result.reason}</small>
            </div>
          )}
          
          {result.status === 'insufficient_evidence' && (
            <div className="insufficient">
              <p>{result.answer}</p>
              <p>该知识点正在完善教材证据中。</p>
            </div>
          )}
        </div>
      )}
    </div>
  );
}
```

```python
def test_normalize_text():
    from qa.service import QAService
    qa = QAService(None)
    
    assert qa._normalize_text("TLB") == qa._normalize_text("ｔｌｂ")
    assert qa._normalize_text("快表") == qa._normalize_text("  快表  ")
    assert "os" not in qa._normalize_text("most")  # 单词边界
```

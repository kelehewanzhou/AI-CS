## 本次只做什么

1. 实现auth模块：Argon2id密码哈希、会话创建/验证/撤销
2. 实现API：POST /auth/login、GET /auth/me、POST /auth/logout、POST /auth/password
3. 实现CLI命令`python cli/accounts.py create`，生成随机临时密码
4. 实现FastAPI依赖`current_user`，验证Cookie、返回UserContext
5. 实现Origin校验（§5.4），拒绝跨域POST

- 不实现SSO或第三方登录
- 不实现邮件找回密码
- 不实现图形验证码
- 不在代码中内置默认账号

### backend/auth/service.py

```python
from argon2 import PasswordHasher
from argon2.exceptions import VerifyMismatchError
import secrets
import hashlib
from datetime import datetime, timedelta, timezone
from sqlalchemy.orm import Session
from common.errors import AppException
from .models import User, Session as SessionModel
from .repository import UserRepository, SessionRepository

ph = PasswordHasher()

class AuthService:
    def __init__(self, db: Session):
        self.db = db
        self.user_repo = UserRepository(db)
        self.session_repo = SessionRepository(db)
    
    def login(self, username: str, password: str) -> tuple[UserContext, str, datetime]:
        """登录，返回用户上下文、令牌、过期时间"""
        user = self.user_repo.get_by_username(username)
        if not user or not user.is_active:
            raise AppException("UNAUTHENTICATED", "账号或密码不正确", 401)
        
        try:
            ph.verify(user.password_hash, password)
        except VerifyMismatchError:
            raise AppException("UNAUTHENTICATED", "账号或密码不正确", 401)
        
        # 生成会话令牌（32字节随机）
        token = secrets.token_urlsafe(32)
        token_hash = hashlib.sha256(token.encode()).hexdigest()
        
        expires_at = datetime.now(timezone.utc) + timedelta(seconds=8*3600)
        
        session = SessionModel(
            user_id=user.user_id,
            token_hash=token_hash,
            expires_at=expires_at
        )
        self.db.add(session)
        self.db.commit()
        
        user_ctx = UserContext(
            user_id=user.user_id,
            username=user.username,
            role=user.role,
            must_change_password=user.must_change_password
        )
        
        return user_ctx, token, expires_at
    
    def verify_token(self, token: str) -> UserContext:
        """验证令牌，返回用户上下文"""
        token_hash = hashlib.sha256(token.encode()).hexdigest()
        session = self.session_repo.get_by_token_hash(token_hash)
        
        if not session:
            raise AppException("UNAUTHENTICATED", "会话无效", 401)
        
        if session.revoked_at:
            raise AppException("UNAUTHENTICATED", "会话已撤销", 401)
        
        if datetime.now(timezone.utc) > session.expires_at:
            raise AppException("UNAUTHENTICATED", "会话已过期", 401)
        
        user = self.user_repo.get_by_id(session.user_id)
        if not user or not user.is_active:
            raise AppException("UNAUTHENTICATED", "用户不存在或已停用", 401)
        
        return UserContext(
            user_id=user.user_id,
            username=user.username,
            role=user.role,
            must_change_password=user.must_change_password
        )
    
    def logout(self, token: str):
        """撤销会话"""
        token_hash = hashlib.sha256(token.encode()).hexdigest()
        session = self.session_repo.get_by_token_hash(token_hash)
        if session and not session.revoked_at:
            session.revoked_at = datetime.now(timezone.utc)
            self.db.commit()
    
    def change_password(self, user_id: UUID, current_password: str, new_password: str):
        """修改密码，撤销该用户所有会话"""
        user = self.user_repo.get_by_id(user_id)
        
        try:
            ph.verify(user.password_hash, current_password)
        except VerifyMismatchError:
            raise AppException("FORBIDDEN", "当前密码不正确", 403)
        
        if current_password == new_password:
            raise AppException("VALIDATION_ERROR", "新密码不能与旧密码相同", 422)
        
        if not (12 <= len(new_password) <= 128):
            raise AppException("VALIDATION_ERROR", "密码长度必须为12-128字符", 422)
        
        user.password_hash = ph.hash(new_password)
        user.must_change_password = False
        user.password_changed_at = datetime.now(timezone.utc)
        
        # 撤销该用户所有会话
        self.session_repo.revoke_all_by_user(user_id)
        
        self.db.commit()
```

### backend/api/routers/auth.py

```python
from fastapi import APIRouter, Depends, Request, Response
from sqlalchemy.orm import Session
from pydantic import BaseModel, Field
from common.database import get_db
from common.errors import AppException
from common.config import settings
from auth.service import AuthService

router = APIRouter()

class LoginRequest(BaseModel):
    username: str = Field(min_length=3, max_length=64)
    password: str = Field(min_length=1, max_length=128)

class PasswordChangeRequest(BaseModel):
    current_password: str = Field(min_length=1, max_length=128)
    new_password: str = Field(min_length=12, max_length=128)

def check_origin(request: Request):
    """检查Origin，POST请求必须"""
    if request.method == "POST":
        origin = request.headers.get("origin")
        allowed = [o.strip() for o in settings.allowed_origins.split(",")]
        if not origin or origin not in allowed:
            raise AppException("FORBIDDEN", "不允许的Origin", 403)

def get_current_user(request: Request, db: Session = Depends(get_db)) -> UserContext:
    """从Cookie提取并验证用户"""
    token = request.cookies.get(settings.session_cookie_name)
    if not token:
        raise AppException("UNAUTHENTICATED", "未登录", 401)
    
    auth_service = AuthService(db)
    return auth_service.verify_token(token)

def require_password_changed(user: UserContext = Depends(get_current_user)):
    """必须已改密"""
    if user.must_change_password:
        raise AppException("FORBIDDEN", "请先修改初始密码", 403)
    return user

@router.post("/login")
def login(
    req: LoginRequest,
    response: Response,
    request: Request,
    db: Session = Depends(get_db)
):
    check_origin(request)
    
    auth_service = AuthService(db)
    user_ctx, token, expires_at = auth_service.login(req.username, req.password)
    
    # 设置Cookie
    response.set_cookie(
        key=settings.session_cookie_name,
        value=token,
        httponly=True,
        samesite="lax",
        secure=(settings.environment != "development"),
        path="/",
        max_age=8*3600
    )
    
    return {
        "data": {
            "user": {
                "user_id": str(user_ctx.user_id),
                "username": user_ctx.username,
                "role": user_ctx.role,
                "must_change_password": user_ctx.must_change_password
            },
            "expires_at": expires_at.isoformat()
        }
    }

@router.get("/me")
def get_me(user: UserContext = Depends(get_current_user)):
    return {
        "data": {
            "user": {
                "user_id": str(user.user_id),
                "username": user.username,
                "role": user.role,
                "must_change_password": user.must_change_password
            }
        }
    }

@router.post("/logout")
def logout(
    request: Request,
    response: Response,
    db: Session = Depends(get_db)
):
    check_origin(request)
    
    token = request.cookies.get(settings.session_cookie_name)
    if token:
        auth_service = AuthService(db)
        auth_service.logout(token)
    
    response.delete_cookie(key=settings.session_cookie_name, path="/")
    
    return {"data": {"logged_out": True}}

@router.post("/password")
def change_password(
    req: PasswordChangeRequest,
    request: Request,
    response: Response,
    user: UserContext = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    check_origin(request)
    
    auth_service = AuthService(db)
    auth_service.change_password(user.user_id, req.current_password, req.new_password)
    
    # 清除Cookie
    response.delete_cookie(key=settings.session_cookie_name, path="/")
    
    return {
        "data": {
            "password_changed": True,
            "reauth_required": True
        }
    }
```

### backend/cli/accounts.py

```python
import argparse
import secrets
import string
from pathlib import Path
from datetime import datetime
from argon2 import PasswordHasher
from sqlalchemy.orm import Session
from common.database import engine
from auth.models import User

ph = PasswordHasher()

def generate_password(length=16):
    """生成随机临时密码"""
    chars = string.ascii_letters + string.digits + "!@#$%"
    return ''.join(secrets.choice(chars) for _ in range(length))

def create_account(username: str, role: str):
    """创建账号"""
    if role not in ['student', 'teacher']:
        print(f"错误：role必须为student或teacher")
        return 1
    
    temp_password = generate_password()
    password_hash = ph.hash(temp_password)
    
    with Session(engine) as db:
        # 检查用户名是否已存在
        existing = db.query(User).filter(User.username == username).first()
        if existing:
            print(f"错误：用户名{username}已存在")
            return 1
        
        user = User(
            username=username,
            password_hash=password_hash,
            role=role,
            must_change_password=True
        )
        db.add(user)
        db.commit()
        
        # 写入凭据文件（权限受限目录）
        output_dir = Path("./tmp")
        output_dir.mkdir(exist_ok=True)
        
        timestamp = datetime.now().strftime("%Y%m%d")
        output_file = output_dir / f"credentials_{timestamp}.txt"
        
        with open(output_file, 'a', encoding='utf-8') as f:
            f.write(f"{username}\t{role}\t{temp_password}\n")
        
        print(f"账号创建成功：{username} ({role})")
        print(f"临时密码已写入：{output_file}")
        print("请通过安全渠道发放，发放后删除该文件")
    
    return 0

def main():
    parser = argparse.ArgumentParser(description="账号管理CLI")
    subparsers = parser.add_subparsers(dest="command")
    
    create_parser = subparsers.add_parser("create", help="创建账号")
    create_parser.add_argument("--username", required=True)
    create_parser.add_argument("--role", required=True, choices=['student', 'teacher'])
    
    args = parser.parse_args()
    
    if args.command == "create":
        return create_account(args.username, args.role)
    else:
        parser.print_help()
        return 1

if __name__ == "__main__":
    exit(main())
```

```markdown
# 认证安全说明

## 密码存储
- 使用Argon2id算法
- argon2-cffi默认参数（适合服务器环境）
- 不存储明文密码

## 会话管理
- 令牌：32字节随机（secrets.token_urlsafe）
- 存储：SHA-256哈希
- 有效期：8小时，不滑动续期
- Cookie：HttpOnly, SameSite=Lax, Secure（生产）

## Origin校验
- POST请求必须验证Origin头
- 允许列表在.env配置
- localhost仅开发环境

## 限流
由Nginx实现（任务9），/auth/login为5次/分钟

## 已知限制
- 无分布式会话存储（单机部署）
- 无账号锁定机制
- 无密码复杂度强制（仅长度）
```

```bash
  echo "取消恢复"
  exit 0

# 创建恢复测试数据库

  -c "DROP DATABASE IF EXISTS kglearn_restore_test;"
psql -h postgres -U kglearn_user -d postgres \
  -c "CREATE DATABASE kglearn_restore_test;"

# 恢复到测试库
pg_restore -h postgres -U kglearn_user \
  -d kglearn_restore_test "${BACKUP_FILE}"

# 验证记录数
psql -h postgres -U kglearn_user -d kglearn_restore_test \
  -c "SELECT 'users' AS table_name, COUNT(*) FROM users

echo "恢复到测试库成功，请验证数据完整性"
echo "确认无误后，使用生产恢复脚本替换主库"
```

### nginx/conf.d/default.conf

```nginx
# 限流区域
limit_req_zone $binary_remote_addr zone=login_limit:10m rate=5r/m;

# 上游
upstream backend {
    server backend:8000 max_fails=3 fail_timeout=30s;
}

# HTTP重定向到HTTPS（生产环境）
server {
    listen 80;
    server_name example.edu.cn;
    
    # ACME challenge路径
    location /.well-known/acme-challenge/ {
        root /var/www/certbot;
    }
    
    location / {
        return 301 https://$host$request_uri;
    }
}

# HTTPS服务器
server {
    listen 443 ssl http2;
    server_name example.edu.cn;
    
    # TLS配置
    ssl_certificate /etc/nginx/ssl/fullchain.pem;
    ssl_certificate_key /etc/nginx/ssl/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;
    
    # 日志（不记录Cookie和密码）
    access_log /var/log/nginx/access.log combined;
    error_log /var/log/nginx/error.log warn;
    
    # 静态文件
    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/index.html =404;
        add_header Cache-Control "no-cache, no-store, must-revalidate";
        add_header Pragma "no-cache";
        add_header Expires "0";
    }
    
    # API代理
    location /api/v1/ {
        proxy_pass http://backend;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Request-ID $request_id;
        
        # 超时
        proxy_connect_timeout 10s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }
    
    # 登录限流
    location /api/v1/auth/login {
        limit_req zone=login_limit burst=5;
        limit_req_status 429;
        
        proxy_pass http://backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Request-ID $request_id;
    }
    
    # 健康检查（无限流）
    location /health/ {
        proxy_pass http://backend;
    }
    
    # 错误页面
    error_page 502 503 504 /50x.json;
    location = /50x.json {
        return 503 '{"code":"UNAVAILABLE","message":"服务暂时不可用","request_id":"nginx"}';
        add_header Content-Type application/json always;
    }
    
    # 安全头
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
}
```

### scripts/deploy.sh

```bash
#!/bin/bash
set -e

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(dirname "$SCRIPT_DIR")"
cd "$PROJECT_ROOT"

# 加载环境变量
if [ -f .env ]; then
    export $(cat .env | grep -v '^#' | xargs)
fi

VERSION=${1:-latest}
echo "部署版本：$VERSION"

# 1. 备份
echo "执行备份..."
docker compose exec postgres /backup/backup.sh

# 2. 构建镜像
echo "构建后端镜像..."
docker build -t kglearn-backend:$VERSION -f backend/Dockerfile .

echo "构建前端镜像..."
cd web && pnpm build && cd ..
docker build -t kglearn-nginx:$VERSION -f nginx/Dockerfile .

# 3. 停止旧服务（保留数据库）
echo "停止旧服务..."
docker compose stop backend nginx

# 4. 启动新服务
echo "启动新服务..."
export VERSION=$VERSION
docker compose up -d migrate
docker compose up -d seed-import
docker compose up -d backend
docker compose up -d nginx

# 5. 健康检查
echo "等待服务就绪..."
sleep 10

if curl -f http://localhost/health/ready; then
    echo "✓ 部署成功"
else
    echo "✗ 健康检查失败，回滚..."
    docker compose logs backend
    exit 1
fi

echo "部署完成：$VERSION"
```

### scripts/backup.sh

```bash
#!/bin/bash
set -e

DATE=$(TZ=Asia/Shanghai date +%Y%m%d_%H%M%S)
BACKUP_DIR="/backup"
BACKUP_FILE="${BACKUP_DIR}/kglearn_${DATE}.dump"

# PostgreSQL custom format
pg_dump -h localhost -U kglearn_user -d kglearn \
  -Fc -f "${BACKUP_FILE}"

# 验证备份
pg_restore --list "${BACKUP_FILE}" > /dev/null

# 清理旧备份（保留14天日备份）
find "${BACKUP_DIR}" -name "kglearn_*.dump" -mtime +14 -type f -delete

# 每周日创建周备份（保留4周）
if [ $(date +%u) -eq 7 ]; then
    WEEK_BACKUP="${BACKUP_DIR}/weekly/kglearn_week_${DATE}.dump"
    mkdir -p "${BACKUP_DIR}/weekly"
    cp "${BACKUP_FILE}" "${WEEK_BACKUP}"
    find "${BACKUP_DIR}/weekly" -name "kglearn_week_*.dump" -mtime +28 -delete
fi

echo "备份完成：${BACKUP_FILE}"
ls -lh "${BACKUP_FILE}"
```
